①ifedge_setting_Fujitsu.json：「socket_fujitsu.js」関連のjsonの説明。
{
    "response_telegram": {
        "socket-type": "fujitsu",						←電子カルテの種類（例：富士通)
        "type_name": "ResponseType",						←応答種別のJOSN設定名：header部分nameを設定した内容
        "length_name": "TelegramLength",					←電文長のJOSN設定名：header部分nameを設定した内容
        "type_name_success": "OK",						←正常応答電文の応答種別の値
        "type_name_failure": "N2",						←異常応答電文の応答種別の値
        "retry_value": "N1",							←再送要求の応答種別の値
        "skip_value": "N2",							←スキップの応答種別の値
        "abnormal_value": ["NG","N3","N4"],					←接続異常の応答種別の値
        "header_length_included": false,					←受信電文の電文長にheader部分の長を含みますか（true：含める、false：含まない）
        "response_success": {							←正常応答電文の定義
            "header_length": 65,						←応答電文の長さ（★header_lengthは数値です）
            "header": [								←応答電文項目
                {
                    "name": "TlegramType",					←応答電文.項目名
                    "length": 2,						←応答電文.長さ（★length数値は数値です）
                    "value": ""							←応答電文.値：空の文字列（受信電文の値)
                },
                {
                    "name": "RecordContinuationInstructions",
                    "length": 1,
                    "value": ""
                },
                {
                    "name": "DestinationSystemCode",
                    "length": 2,
                    "value": "XX"						←応答電文.値：固定値文字列（例：XX)
                },
                {
                    "name": "SourceSystemCode",
                    "length": 2,
                    "value": "VN"
                },
                {
                    "name": "DATE",
                    "length": 8,
                    "value": "$DATE",						←応答電文.値：日付（例：XX)
                    "format": "YYYYMMDD"					←応答電文.書式（例：YYYYMMDD)
                },
                {
                    "name": "TIME",
                    "length": 6,
                    "value": "$DATE",
                    "format": "HH24MISS"

                {

                    "name": "ResponseType",
                    "length": 2,
                    "value": "$TYPENAME"					←応答電文.値：応答種別の値($TYPENAME:type_name_successとtype_name_failureの値)
                },
                {
                    "name": "TelegramLength",
